cmake_minimum_required(VERSION 3.24)
project(jpi_jvmti LANGUAGES CXX)
option(JPI_BUILD_TEST_FIXTURES "Build Linux lifecycle fixtures outside packaged resources" OFF)
find_package(JNI REQUIRED COMPONENTS JVM)
add_library(jpi_jvmti SHARED agent/jpi_jvmti.cpp agent/capture_archive.cpp memory/raw_memory.cpp)
find_package(Threads REQUIRED)
target_link_libraries(jpi_jvmti PRIVATE Threads::Threads)
target_link_libraries(jpi_jvmti PRIVATE ${CMAKE_DL_LIBS})
target_include_directories(jpi_jvmti PRIVATE ${JNI_INCLUDE_DIRS})
target_compile_features(jpi_jvmti PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(jpi_jvmti PRIVATE /W4 /WX /EHsc)
    set_property(TARGET jpi_jvmti PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
    target_compile_options(jpi_jvmti PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
    if(MINGW)
        target_link_options(jpi_jvmti PRIVATE -static-libgcc -static-libstdc++)
    endif()
endif()
if(WIN32)
    set(JPI_OS windows)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(JPI_OS linux)
else()
    message(FATAL_ERROR "The native backend currently supports Windows and Linux")
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 8 AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64|X86_64)$")
    set(JPI_ARCH x86_64)
elseif(CMAKE_SIZEOF_VOID_P EQUAL 8 AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
    set(JPI_ARCH aarch64)
else()
    message(FATAL_ERROR "Unsupported target architecture")
endif()
if(NOT JPI_NATIVE_OUTPUT_DIR)
    set(JPI_NATIVE_OUTPUT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../target/native-resources")
endif()
get_filename_component(JPI_NATIVE_OUTPUT_DIR "${JPI_NATIVE_OUTPUT_DIR}" ABSOLUTE)
set(JPI_OUTPUT "${JPI_NATIVE_OUTPUT_DIR}/native/${JPI_OS}-${JPI_ARCH}")
set_target_properties(jpi_jvmti PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${JPI_OUTPUT}" LIBRARY_OUTPUT_DIRECTORY "${JPI_OUTPUT}")
foreach(JPI_CONFIG DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(jpi_jvmti PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${JPI_CONFIG} "${JPI_OUTPUT}" LIBRARY_OUTPUT_DIRECTORY_${JPI_CONFIG} "${JPI_OUTPUT}")
endforeach()
add_custom_command(TARGET jpi_jvmti POST_BUILD COMMAND ${CMAKE_COMMAND} "-DJPI_LIBRARY=$<TARGET_FILE:jpi_jvmti>" -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/write-sha256.cmake")

if(WIN32 AND JPI_ARCH STREQUAL "x86_64")
    add_library(process_injector SHARED injectors/windows/process_injector_windows.cpp)
    target_link_libraries(process_injector PRIVATE psapi)
    target_include_directories(process_injector PRIVATE ${JNI_INCLUDE_DIRS})
    target_compile_features(process_injector PRIVATE cxx_std_17)
    if(MSVC)
        target_compile_options(process_injector PRIVATE /W4 /WX /EHsc)
        set_property(TARGET process_injector PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    else()
        target_compile_options(process_injector PRIVATE -Wall -Wextra -Werror)
    endif()
    set_target_properties(process_injector PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${JPI_OUTPUT}")
    foreach(JPI_CONFIG DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
        set_target_properties(process_injector PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${JPI_CONFIG} "${JPI_OUTPUT}")
    endforeach()
    add_custom_command(TARGET process_injector POST_BUILD COMMAND ${CMAKE_COMMAND} "-DJPI_LIBRARY=$<TARGET_FILE:process_injector>" -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/write-sha256.cmake")
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND JPI_ARCH STREQUAL "x86_64")
    enable_language(ASM)
    add_library(process_injector SHARED injectors/linux/process_injector_linux.cpp injectors/linux/linux_worker.S)
    target_include_directories(process_injector PRIVATE ${JNI_INCLUDE_DIRS})
    target_compile_features(process_injector PRIVATE cxx_std_17)
    target_compile_options(process_injector PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Werror;-fvisibility=hidden>)
    target_link_libraries(process_injector PRIVATE Threads::Threads)
    set_target_properties(process_injector PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${JPI_OUTPUT}")
    foreach(JPI_CONFIG DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
        set_target_properties(process_injector PROPERTIES LIBRARY_OUTPUT_DIRECTORY_${JPI_CONFIG} "${JPI_OUTPUT}")
    endforeach()
    add_custom_command(TARGET process_injector POST_BUILD COMMAND ${CMAKE_COMMAND} "-DJPI_LIBRARY=$<TARGET_FILE:process_injector>" -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/write-sha256.cmake")
    add_custom_command(TARGET jpi_jvmti POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy "$<TARGET_FILE:jpi_jvmti>" "${JPI_OUTPUT}/libjvmti_agent.so"
        COMMAND ${CMAKE_COMMAND} "-DJPI_LIBRARY=${JPI_OUTPUT}/libjvmti_agent.so" -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/write-sha256.cmake")
    if(JPI_BUILD_TEST_FIXTURES)
        enable_testing()
        add_executable(linux_tracee_test tests/linux_tracee_test.cpp injectors/linux/linux_worker.S)
        target_include_directories(linux_tracee_test PRIVATE ${JNI_INCLUDE_DIRS})
        target_compile_features(linux_tracee_test PRIVATE cxx_std_17)
        target_compile_options(linux_tracee_test PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Werror>)
        target_link_libraries(linux_tracee_test PRIVATE Threads::Threads)
        foreach(JPI_CASE state usr1 trap timeout denied breakpoint)
            add_test(NAME recovery_${JPI_CASE} COMMAND linux_tracee_test ${JPI_CASE})
            set_tests_properties(recovery_${JPI_CASE} PROPERTIES TIMEOUT 15)
        endforeach()
        foreach(JPI_FIXTURE entry_error entry_slow entry_release)
            add_library(jpi_test_${JPI_FIXTURE} SHARED tests/agent_entry_fixture.cpp)
            target_compile_features(jpi_test_${JPI_FIXTURE} PRIVATE cxx_std_17)
            target_compile_options(jpi_test_${JPI_FIXTURE} PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
            target_link_libraries(jpi_test_${JPI_FIXTURE} PRIVATE Threads::Threads)
            set_target_properties(jpi_test_${JPI_FIXTURE} PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../target/native-test-fixtures")
        endforeach()
        target_compile_definitions(jpi_test_entry_slow PRIVATE JPI_TEST_DELAY=20)
        target_compile_definitions(jpi_test_entry_release PRIVATE JPI_TEST_RELEASE=1)
    endif()
endif()
